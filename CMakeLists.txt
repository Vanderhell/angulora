cmake_minimum_required(VERSION 3.20)
project(angulora VERSION 0.1.0 LANGUAGES C)
option(ANGULORA_BUILD_TESTS "Build migrated tests" ON)
option(ANGULORA_STRICT_WARNINGS "Treat compiler warnings as errors" ON)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
file(GLOB ANGULORA_SOURCES CONFIGURE_DEPENDS src/*.c)
add_library(angulora STATIC ${ANGULORA_SOURCES})
target_include_directories(angulora PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/angulora/trigonometry)
target_compile_features(angulora PUBLIC c_std_99)
if(ANGULORA_STRICT_WARNINGS)
    if(MSVC)
        target_compile_options(angulora PRIVATE /W4 /WX)
    else()
        target_compile_options(angulora PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
if(ANGULORA_BUILD_TESTS)
 enable_testing()
 set(ANGULORA_TESTS tests/test_audit.c)
 foreach(test_source IN LISTS ANGULORA_TESTS)
  get_filename_component(test_name "${test_source}" NAME_WE)
  add_executable(angulora_${test_name} ${test_source})
  target_include_directories(angulora_${test_name} PRIVATE
      ${CMAKE_CURRENT_SOURCE_DIR}/include
      ${CMAKE_CURRENT_SOURCE_DIR}/include/angulora/trigonometry)
  target_link_libraries(angulora_${test_name} PRIVATE angulora)
  if(ANGULORA_STRICT_WARNINGS)
   if(MSVC)
    target_compile_options(angulora_${test_name} PRIVATE /W4 /WX)
   else()
    target_compile_options(angulora_${test_name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
   endif()
  endif()
  if(NOT MSVC)
   target_link_libraries(angulora_${test_name} PRIVATE m)
  endif()
  add_test(NAME "${test_name}" COMMAND "angulora_${test_name}")
 endforeach()
endif()
